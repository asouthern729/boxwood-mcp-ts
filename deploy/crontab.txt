# ams360-demo scheduled jobs — the lines to add to andrew's crontab. UTC-pair + runtime TZ check
# because this box's cron doesn't support CRON_TZ. The demo DB reseed (08:05 Central, owned by
# postgres-mcp/demo) is a separate line; the morning download runs at 08:10 so it reads the fresh seed.
10 13,14 * * 1-5 [ "$(TZ=America/Chicago date +\%H:\%M)" = "08:10" ] && /opt/containers/ams360-demo/run-job.sh morning-download
15 13,14 * * * [ "$(TZ=America/Chicago date +\%H:\%M)" = "08:15" ] && find /var/lib/ams360-demo/output/download-report -maxdepth 1 -type f -name '[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]_download_report.xlsx' -mtime +30 -delete
30 * 1,2 * * [ "$(TZ=America/Chicago date +\%d-\%H:\%M)" = "01-08:30" ] && /opt/containers/ams360-demo/run-job.sh monthly
0 14,15 * * 1-5 [ "$(TZ=America/Chicago date +\%H:\%M)" = "09:00" ] && /opt/containers/ams360-demo/run-job.sh renewal-premium-summary-refresh
30 13,14 * * 1-5 [ "$(TZ=America/Chicago date +\%H:\%M)" = "08:30" ] && /opt/containers/ams360-demo/run-job.sh premium-change
0 14,15 * * 1-5 [ "$(TZ=America/Chicago date +\%H:\%M)" = "09:00" ] && /opt/containers/ams360-demo/run-job.sh renewal-summary
0 14,15 * * 1-5 [ "$(TZ=America/Chicago date +\%H:\%M)" = "09:00" ] && /opt/containers/ams360-demo/run-job.sh cl-renewal-summaries
